(define_predicate "s_register_operand"
  (match_code "reg,subreg")
{
  int regno;

  if (GET_CODE (op) == SUBREG)
    op = SUBREG_REG (op);

  if (GET_CODE (op) != REG)
    return 0;

  regno = REGNO (op);
  if (regno >= FIRST_PSEUDO_REGISTER)
    {
      if (!TC32_REG_STRICT_P)
        return 1;

      regno = reg_renumber[regno];
      if (regno < 0)
        return 0;
    }

  return REGNO_REG_CLASS (regno) != NO_REGS;
})

(define_predicate "low_register_operand"
  (match_code "reg,subreg")
{
  int regno;

  if (GET_CODE (op) == SUBREG)
    op = SUBREG_REG (op);

  if (GET_CODE (op) != REG)
    return 0;

  regno = REGNO (op);
  if (regno >= FIRST_PSEUDO_REGISTER)
    {
      if (!TC32_REG_STRICT_P)
        return 1;

      regno = reg_renumber[regno];
      if (regno < 0)
        return 0;
    }

  return regno <= LAST_LO_REGNUM;
})

(define_predicate "hard_low_register_operand"
  (match_code "reg,subreg")
{
  if (GET_CODE (op) == SUBREG)
    op = SUBREG_REG (op);

  return (GET_CODE (op) == REG
          && REGNO (op) < FIRST_PSEUDO_REGISTER
          && REGNO (op) <= LAST_LO_REGNUM);
})

(define_predicate "core_register_operand"
  (match_code "reg,subreg")
{
  int regno;

  if (GET_CODE (op) == SUBREG)
    op = SUBREG_REG (op);

  if (GET_CODE (op) != REG)
    return 0;

  regno = REGNO (op);
  if (regno >= FIRST_PSEUDO_REGISTER)
    {
      if (!TC32_REG_STRICT_P)
        return 1;

      regno = reg_renumber[regno];
      if (regno < 0)
        return 0;
    }

  return (regno <= LAST_LO_REGNUM
          || (regno >= 8 && regno <= 11));
})

(define_predicate "hard_core_register_operand"
  (match_code "reg,subreg")
{
  if (GET_CODE (op) == SUBREG)
    op = SUBREG_REG (op);

  return (GET_CODE (op) == REG
          && REGNO (op) < FIRST_PSEUDO_REGISTER
          && (REGNO (op) <= LAST_LO_REGNUM
              || (REGNO (op) >= 8 && REGNO (op) <= 11)));
})

(define_predicate "reg_or_int_operand"
  (ior (match_code "const_int")
       (match_operand 0 "s_register_operand")))

(define_predicate "low_reg_or_int_operand"
  (ior (and (match_code "const_int")
            (match_test "INTVAL (op) >= 0 && INTVAL (op) <= 255"))
       (match_operand 0 "low_register_operand")))

(define_predicate "low_reg_or_addsub_int_operand"
  (ior (match_code "const_int")
       (match_operand 0 "low_register_operand")))

(define_predicate "stack_pointer_operand"
  (and (match_code "reg")
       (match_test "REGNO (op) == STACK_POINTER_REGNUM")))

(define_predicate "special_register_operand"
  (and (match_code "reg")
       (match_test "REGNO (op) == FRAME_POINTER_REGNUM
                    || REGNO (op) == ARG_POINTER_REGNUM")))

(define_predicate "special_neg_mem_operand"
  (match_code "mem")
{
  rtx addr = XEXP (op, 0);

  return (GET_CODE (addr) == PLUS
          && GET_CODE (XEXP (addr, 0)) == REG
          && (REGNO (XEXP (addr, 0)) == STACK_POINTER_REGNUM
              || REGNO (XEXP (addr, 0)) == FRAME_POINTER_REGNUM
              || REGNO (XEXP (addr, 0)) == ARG_POINTER_REGNUM)
          && GET_CODE (XEXP (addr, 1)) == CONST_INT
          && INTVAL (XEXP (addr, 1)) < 0
          && INTVAL (XEXP (addr, 1)) > -256);
})

(define_predicate "special_large_neg_mem_operand"
  (match_code "mem")
{
  rtx addr = XEXP (op, 0);

  return (GET_CODE (addr) == PLUS
          && GET_CODE (XEXP (addr, 0)) == REG
          && (REGNO (XEXP (addr, 0)) == STACK_POINTER_REGNUM
              || REGNO (XEXP (addr, 0)) == FRAME_POINTER_REGNUM
              || REGNO (XEXP (addr, 0)) == ARG_POINTER_REGNUM)
          && GET_CODE (XEXP (addr, 1)) == CONST_INT
          && INTVAL (XEXP (addr, 1)) <= -256);
})

(define_special_predicate "multi_register_push"
  (match_code "parallel")
{
  if ((GET_CODE (XVECEXP (op, 0, 0)) != SET)
      || (GET_CODE (SET_SRC (XVECEXP (op, 0, 0))) != UNSPEC)
      || (XINT (SET_SRC (XVECEXP (op, 0, 0)), 1) != UNSPEC_PUSH_MULT))
    return false;

  return true;
})

(define_expand "movsi"
  [(set (match_operand:SI 0 "general_operand" "")
        (match_operand:SI 1 "general_operand" ""))]
  ""
{
  if (GET_CODE (operands[1]) == REG
      && REGNO (operands[1]) == ARG_POINTER_REGNUM
      && can_create_pseudo_p ())
    {
      int base_regno = (frame_pointer_needed
                        ? HARD_FRAME_POINTER_REGNUM : STACK_POINTER_REGNUM);
      HOST_WIDE_INT offset = tc32_initial_elimination_offset (ARG_POINTER_REGNUM,
                                                              base_regno);
      rtx tmp = gen_reg_rtx (SImode);

      emit_move_insn (tmp, gen_rtx_REG (SImode, base_regno));
      if (offset != 0)
        emit_insn (gen_addsi3 (tmp, tmp, GEN_INT (offset)));

      if (GET_CODE (operands[0]) == MEM)
        {
          emit_move_insn (operands[0], tmp);
          DONE;
        }

      operands[1] = tmp;
    }

  if (can_create_pseudo_p ())
    {
      if (GET_CODE (operands[0]) == MEM
          && !memory_address_p (GET_MODE (operands[0]), XEXP (operands[0], 0)))
        operands[0]
          = replace_equiv_address (operands[0],
                                   force_operand (XEXP (operands[0], 0), NULL_RTX));

      if (GET_CODE (operands[1]) == MEM
          && !memory_address_p (GET_MODE (operands[1]), XEXP (operands[1], 0)))
        operands[1]
          = replace_equiv_address (operands[1],
                                   force_operand (XEXP (operands[1], 0), NULL_RTX));
    }

  if (GET_CODE (operands[0]) == MEM && GET_CODE (operands[1]) == MEM)
    operands[1] = force_reg (SImode, operands[1]);

  if (GET_CODE (operands[0]) == MEM && GET_CODE (operands[1]) != REG)
    operands[1] = force_reg (SImode, operands[1]);

  if (GET_CODE (operands[0]) == MEM
      && GET_CODE (operands[1]) == REG
      && REGNO (operands[1]) < FIRST_PSEUDO_REGISTER
      && REGNO (operands[1]) > LAST_LO_REGNUM)
    {
      if (can_create_pseudo_p ())
        operands[1] = copy_to_mode_reg (SImode, operands[1]);
    }

  if (GET_CODE (operands[0]) == REG
      && REGNO (operands[0]) < FIRST_PSEUDO_REGISTER
      && REGNO (operands[0]) > LAST_LO_REGNUM
      && GET_CODE (operands[1]) == MEM)
    {
      if (can_create_pseudo_p ())
        {
          rtx tmp = gen_reg_rtx (SImode);
          emit_move_insn (tmp, operands[1]);
          emit_move_insn (operands[0], tmp);
          DONE;
        }
    }

  if (GET_CODE (operands[0]) == REG
      && (GET_CODE (operands[1]) == SYMBOL_REF
          || GET_CODE (operands[1]) == LABEL_REF
          || GET_CODE (operands[1]) == CONST
          || (CONSTANT_P (operands[1])
              && GET_CODE (operands[1]) != CONST_INT)))
    operands[1] = force_const_mem (SImode, operands[1]);

  if (GET_CODE (operands[0]) == REG
      && GET_CODE (operands[1]) == CONST_INT
      && (INTVAL (operands[1]) < -255
          || INTVAL (operands[1]) > 255))
    operands[1] = force_const_mem (SImode, operands[1]);
})

(define_expand "cstoresi4"
  [(set (match_operand:SI 0 "register_operand" "")
        (match_operator:SI 1 "comparison_operator"
         [(match_operand:SI 2 "register_operand" "")
          (match_operand:SI 3 "nonmemory_operand" "")]))]
  ""
{
  rtx scratch, scratch2, op3;

  if (GET_CODE (operands[1]) == LT
      || GET_CODE (operands[1]) == LE
      || GET_CODE (operands[1]) == GT
      || GET_CODE (operands[1]) == GE)
    {
      rtx rhs = operands[3];
      rtx l_true = gen_label_rtx ();
      rtx l_end = gen_label_rtx ();

      if (!register_operand (rhs, SImode) && !CONST_INT_P (rhs))
        rhs = force_reg (SImode, rhs);

      emit_move_insn (operands[0], const0_rtx);
      emit_jump_insn (gen_cbranchsi4 (gen_rtx_fmt_ee (GET_CODE (operands[1]),
                                                      VOIDmode,
                                                      operands[2],
                                                      rhs),
                                      operands[2], rhs, l_true));
      emit_jump_insn (gen_jump (l_end));
      emit_label (l_true);
      emit_move_insn (operands[0], const1_rtx);
      emit_label (l_end);
      DONE;
    }

  if (operands[3] == const0_rtx)
    {
      switch (GET_CODE (operands[1]))
        {
        case EQ:
          emit_insn (gen_cstoresi_eq0_thumb1 (operands[0], operands[2]));
          DONE;

        case NE:
          emit_insn (gen_cstoresi_ne0_thumb1 (operands[0], operands[2]));
          DONE;

        case LT:
          scratch = gen_reg_rtx (SImode);
          emit_insn (gen_ashrsi3 (scratch, operands[2], GEN_INT (31)));
          emit_insn (gen_andsi3 (operands[0], scratch, const1_rtx));
          DONE;

        case GE:
          scratch = gen_reg_rtx (SImode);
          emit_insn (gen_ashrsi3 (scratch, operands[2], GEN_INT (31)));
          emit_insn (gen_one_cmplsi2 (scratch, scratch));
          emit_insn (gen_andsi3 (operands[0], scratch, const1_rtx));
          DONE;

        case LE:
          scratch = expand_binop (SImode, add_optab, operands[2], constm1_rtx,
                                  NULL_RTX, 0, OPTAB_WIDEN);
          scratch = expand_binop (SImode, ior_optab, operands[2], scratch,
                                  NULL_RTX, 0, OPTAB_WIDEN);
          expand_binop (SImode, lshr_optab, scratch, GEN_INT (31),
                        operands[0], 1, OPTAB_WIDEN);
          DONE;

        case GT:
          scratch = expand_binop (SImode, ashr_optab, operands[2], GEN_INT (31),
                                  NULL_RTX, 0, OPTAB_WIDEN);
          scratch = expand_binop (SImode, sub_optab, scratch, operands[2],
                                  NULL_RTX, 0, OPTAB_WIDEN);
          expand_binop (SImode, lshr_optab, scratch, GEN_INT (31),
                        operands[0], 0, OPTAB_WIDEN);
          DONE;

        default:
          break;
        }
    }

  scratch = expand_binop (SImode, sub_optab, operands[2], operands[3],
                          NULL_RTX, 0, OPTAB_WIDEN);
  if (scratch)
    {
      if (GET_CODE (operands[1]) == EQ)
        {
          rtx l_true = gen_label_rtx ();
          rtx l_end = gen_label_rtx ();

          emit_move_insn (operands[0], const0_rtx);
          emit_jump_insn (gen_cbranchsi4 (gen_rtx_EQ (VOIDmode, scratch, const0_rtx),
                                          scratch, const0_rtx, l_true));
          emit_jump_insn (gen_jump (l_end));
          emit_label (l_true);
          emit_move_insn (operands[0], const1_rtx);
          emit_label (l_end);
          DONE;
        }
      else if (GET_CODE (operands[1]) == NE)
        {
          rtx l_true = gen_label_rtx ();
          rtx l_end = gen_label_rtx ();

          emit_move_insn (operands[0], const0_rtx);
          emit_jump_insn (gen_cbranchsi4 (gen_rtx_NE (VOIDmode, scratch, const0_rtx),
                                          scratch, const0_rtx, l_true));
          emit_jump_insn (gen_jump (l_end));
          emit_label (l_true);
          emit_move_insn (operands[0], const1_rtx);
          emit_label (l_end);
          DONE;
        }
    }

  switch (GET_CODE (operands[1]))
    {
    case LE:
      op3 = force_reg (SImode, operands[3]);
      scratch = expand_binop (SImode, lshr_optab, operands[2], GEN_INT (31),
                              NULL_RTX, 1, OPTAB_WIDEN);
      scratch2 = expand_binop (SImode, ashr_optab, op3, GEN_INT (31),
                               NULL_RTX, 0, OPTAB_WIDEN);
      emit_insn (gen_thumb1_addsi3_addgeu (operands[0], scratch, scratch2,
                                           op3, operands[2]));
      DONE;

    case GE:
      op3 = operands[3];
      if (!thumb1_cmp_operand (op3, SImode))
        op3 = force_reg (SImode, op3);
      scratch = expand_binop (SImode, ashr_optab, operands[2], GEN_INT (31),
                              NULL_RTX, 0, OPTAB_WIDEN);
      scratch2 = expand_binop (SImode, lshr_optab, op3, GEN_INT (31),
                               NULL_RTX, 1, OPTAB_WIDEN);
      emit_insn (gen_thumb1_addsi3_addgeu (operands[0], scratch, scratch2,
                                           operands[2], op3));
      DONE;

    case LEU:
      op3 = force_reg (SImode, operands[3]);
      scratch = force_reg (SImode, const0_rtx);
      emit_insn (gen_thumb1_addsi3_addgeu (operands[0], scratch, scratch,
                                           op3, operands[2]));
      DONE;

    case GEU:
      op3 = operands[3];
      if (!thumb1_cmp_operand (op3, SImode))
        op3 = force_reg (SImode, op3);
      scratch = force_reg (SImode, const0_rtx);
      emit_insn (gen_thumb1_addsi3_addgeu (operands[0], scratch, scratch,
                                           operands[2], op3));
      DONE;

    case LTU:
      op3 = operands[3];
      if (!thumb1_cmp_operand (op3, SImode))
        op3 = force_reg (SImode, op3);
      emit_insn (gen_cstoresi_ltu_thumb1 (operands[0], operands[2], op3));
      DONE;

    case GTU:
      op3 = force_reg (SImode, operands[3]);
      emit_insn (gen_cstoresi_ltu_thumb1 (operands[0], op3, operands[2]));
      DONE;

    default:
      break;
    }

  {
    rtx rhs = operands[3];
    rtx l_true = gen_label_rtx ();
    rtx l_end = gen_label_rtx ();

    if (!register_operand (rhs, SImode) && !CONST_INT_P (rhs))
      rhs = force_reg (SImode, rhs);

    emit_move_insn (operands[0], const0_rtx);
    emit_jump_insn (gen_cbranchsi4 (gen_rtx_fmt_ee (GET_CODE (operands[1]),
                                                    VOIDmode,
                                                    operands[2],
                                                    rhs),
                                    operands[2], rhs, l_true));
    emit_jump_insn (gen_jump (l_end));
    emit_label (l_true);
    emit_move_insn (operands[0], const1_rtx);
    emit_label (l_end);
    DONE;
  }
})
